def init_storage(force: bool = False):
    global storage_key
    if storage_key and not force:
        return storage_key
    resp = requests.post(f"{STORAGE_URL}/init", json={"emergent_key": EMERGENT_KEY}, timeout=30)
    resp.raise_for_status()
    storage_key = resp.json()["storage_key"]
    return storage_key


def put_object(path: str, data: bytes, content_type: str) -> dict:
    key = init_storage()
    resp = requests.put(f"{STORAGE_URL}/objects/{path}",
                        headers={"X-Storage-Key": key, "Content-Type": content_type},
                        data=data, timeout=120)
    if resp.status_code == 404:
        key = init_storage(force=True)
        resp = requests.put(f"{STORAGE_URL}/objects/{path}",
                            headers={"X-Storage-Key": key, "Content-Type": content_type},
                            data=data, timeout=120)
    resp.raise_for_status()
    return resp.json()


def get_object(path: str) -> bytes:
    key = init_storage()
    resp = requests.get(f"{STORAGE_URL}/objects/{path}", headers={"X-Storage-Key": key}, timeout=60)
    if resp.status_code == 404:
        key = init_storage(force=True)
        resp = requests.get(f"{STORAGE_URL}/objects/{path}", headers={"X-Storage-Key": key}, timeout=60)
    resp.raise_for_status()
    return resp.content

def session_key(book_id=None, subject=None, topic=None):
    if book_id:
        return f"book-{book_id}"
    if subject and topic:
        return "lesson-" + hashlib.sha256(f"{subject}|{topic}".encode()).hexdigest()[:12]
    return "general"


async def record_activity(subject, topic, is_question):
    now = datetime.now(timezone.utc).isoformat()
    update = {"$set": {"last_active": now}, "$setOnInsert": {"started_at": now}}
    if is_question:
        update["$inc"] = {"question_count": 1}
    else:
        update["$set"]["explained"] = True
    await db.progress.update_one({"subject": subject, "topic": topic}, update, upsert=True)

@api_router.post("/books/upload", response_model=Book)
async def upload_book(file: UploadFile = File(...), lang: str = Form("az")):
    if not file.filename.lower().endswith(".pdf"):
        raise HTTPException(400, "Yalnız PDF faylları qəbul olunur")
    data = await file.read()
    path = f"{APP_NAME}/uploads/{uuid.uuid4()}.pdf"
    try:
        put_object(path, data, "application/pdf")
    except Exception as e:
        logger.error(f"Storage upload failed: {e}")
        raise HTTPException(500, "Fayl yüklənə bilmədi")

    # Analyze PDF with Gemini
    tmp = tempfile.NamedTemporaryFile(delete=False, suffix=".pdf")
    tmp.write(data)
    tmp.close()
    analysis = {}
    try:
        chat = make_chat(f"analyze-{uuid.uuid4()}",
                         "Sən təhsil məzmununu təhlil edən bir köməkçisən. Yalnız JSON qaytar.")
        prompt = (
            "Bu PDF dərsliyi təhlil et və 11-14 yaşlı uşaq üçün sadələşdir. "
            "SADƏCƏ bu formatda düzgün JSON qaytar (başqa mətn yox), dəyərlər "
            + ("Azərbaycan dilində" if lang == "az" else "Türkçe") + " olsun:\n"
            '{"title": "qısa başlıq", "subject": "fənn", "grade": "sinif", '
            '"summary": "2-3 cümləlik xülasə", "key_topics": ["mövzu1","mövzu2","mövzu3"], '
            '"explanation": "uşağa uyğun sadə və maraqlı izah, bənzətmələrlə, 300-500 söz"}'
        )
        pdf_file = FileContentWithMimeType(file_path=tmp.name, mime_type="application/pdf")
        buf = ""
        async for ev in chat.stream_message(UserMessage(text=prompt, file_contents=[pdf_file])):
            if isinstance(ev, TextDelta):
                buf += ev.content
            elif isinstance(ev, StreamDone):
                break
        m = re.search(r"\{.*\}", buf, re.DOTALL)
        if m:
            analysis = json.loads(m.group(0))
    except Exception as e:
        logger.error(f"PDF analysis failed: {e}")
    finally:
        try:
            os.unlink(tmp.name)
        except Exception:
            pass

    book = Book(
        title=analysis.get("title") or file.filename.replace(".pdf", ""),
        subject=analysis.get("subject", ""),
        grade=analysis.get("grade", ""),
        summary=analysis.get("summary", ""),
        key_topics=analysis.get("key_topics", []) or [],
        explanation=analysis.get("explanation", ""),
        storage_path=path,
    )
    doc = book.model_dump()
    doc["is_deleted"] = False
    await db.books.insert_one(doc)
    return book


@api_router.post("/chat")
async def chat_stream(req: ChatRequest):
    book = None
    if req.book_id:
        book = await db.books.find_one({"id": req.book_id})
    session_id = session_key(req.book_id, req.subject, req.topic)
    sysmsg = tutor_system_message(req.lang, book, req.subject, req.topic)

    history = await db.messages.find({"session_id": session_id}).sort("created_at", 1).to_list(20)
    chat = make_chat(session_id, sysmsg)

    if req.explain:
        await db.messages.delete_many({"session_id": session_id})
        user_text = (f"«{req.topic}» mövzusunu 5-ci sinif şagirdinə uyğun, maraqlı və sadə "
                     f"şəkildə, gündəlik nümunələrlə izah et. Qısa və aydın saxla. "
                     f"Sonda şagirdə de ki, indi növbə onundur və bu mövzunu öz sözləri ilə danışmalıdır.")
    elif req.stage == "recall":
        if req.message:
            await db.messages.insert_one({
                "id": str(uuid.uuid4()), "session_id": session_id, "role": "user",
                "content": req.message, "created_at": datetime.now(timezone.utc).isoformat(),
            })
        user_text = (f"Şagird «{req.topic}» mövzusunu öz sözləri ilə belə izah etdi:\n«{req.message}»\n"
                     f"Doğru və yaxşı hissələri qısaca tərif et, varsa səhvləri çox mehribanca düzəlt və "
                     f"düzgün formasını de. Sonra ona de: «Əla! İndi sənə bu mövzudan bir neçə sual verəcəm, hazırsan?»")
    elif req.stage == "quiz_next":
        user_text = (f"Şagirdə «{req.topic}» mövzusundan BİR qısa və sadə sual ver. "
                     f"Əvvəl verdiyin sualları təkrar etmə. Yalnız sualı yaz, cavabı yazma.")
    elif req.stage == "quiz_answer":
        if req.message:
            await db.messages.insert_one({
                "id": str(uuid.uuid4()), "session_id": session_id, "role": "user",
                "content": req.message, "created_at": datetime.now(timezone.utc).isoformat(),
            })
        user_text = (f"Bu, verdiyin sonuncu suala şagirdin cavabıdır:\n«{req.message}»\n"
                     f"Əgər cavab doğrudursa, onu təriflə və dərhal növbəti qısa sualı ver. "
                     f"Əgər cavab səhvdirsə və ya şagird bilmədiyini deyirsə, həmin mövzunu müəllimdən daha "
                     f"yaxşı, daha sadə şəkildə, gündəlik nümunə ilə yenidən izah et, sonra asan bir sual ver. "
                     f"Təxminən 3 sualdan sonra dərsi ürəkləndirici sözlərlə bitir.")
    else:
        user_text = req.message
        await db.messages.insert_one({
            "id": str(uuid.uuid4()), "session_id": session_id, "role": "user",
            "content": req.message, "created_at": datetime.now(timezone.utc).isoformat(),
        })

    context = ""
    if history and not req.explain:
        recent = history[-8:]
        context = "Əvvəlki söhbət:\n" + "\n".join(
            f"{'Şagird' if m['role']=='user' else 'Müəllim'}: {m['content']}" for m in recent) + "\n\nTapşırıq: "

    if req.subject and req.topic:
        a_subject, a_topic = req.subject, req.topic
    elif book:
        a_subject, a_topic = "Yüklənən kitablar", book.get("title", "Kitab")
    else:
        a_subject, a_topic = "Ümumi söhbət", "Suallar"
    await record_activity(a_subject, a_topic, (not req.explain) and bool(req.message))

    async def gen():
        full = ""
        errored = False
        try:
            async for ev in chat.stream_message(UserMessage(text=context + user_text)):
                if isinstance(ev, TextDelta):
                    full += ev.content
                    yield f"data: {json.dumps({'delta': ev.content})}\n\n"
                elif isinstance(ev, StreamDone):
                    break
        except Exception as e:
            logger.error(f"Chat error: {e}")
            errored = True
        if errored or not full.strip():
            yield f"data: {json.dumps({'error': True})}\n\n"
            yield f"data: {json.dumps({'done': True})}\n\n"
            return
        await db.messages.insert_one({
            "id": str(uuid.uuid4()), "session_id": session_id, "role": "assistant",
            "content": full, "created_at": datetime.now(timezone.utc).isoformat(),
        })
        yield f"data: {json.dumps({'done': True})}\n\n"

    return StreamingResponse(gen(), media_type="text/event-stream",
                             headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"})

@api_router.get("/progress")
async def get_progress():
    docs = await db.progress.find({}).sort("last_active", -1).to_list(1000)
    by_subject = {}
    total_q = 0
    for d in docs:
        s = d["subject"]
        by_subject.setdefault(s, {"subject": s, "lessons": [], "total_questions": 0, "last_active": d.get("last_active")})
        by_subject[s]["lessons"].append({
            "topic": d["topic"], "question_count": d.get("question_count", 0),
            "explained": d.get("explained", False), "last_active": d.get("last_active"),
        })
        by_subject[s]["total_questions"] += d.get("question_count", 0)
        total_q += d.get("question_count", 0)
    return {
        "total_questions": total_q,
        "subjects_studied": len(by_subject),
        "lessons_started": len(docs),
        "last_active": docs[0].get("last_active") if docs else None,
        "by_subject": list(by_subject.values()),
    }


def clean_for_tts(text: str) -> str:
    text = re.sub(r"https?://\S+", "", text)
    text = re.sub(r"`{1,3}[^`]*`{1,3}", "", text)
    # speech-prep: verbalize common math for natural Azerbaijani speech
    text = re.sub(r"(\d+)\s*%", r"\1 faiz", text)
    text = re.sub(r"(\d+)\s*/\s*(\d+)", r"\1 bölü \2", text)
    text = text.replace("×", " vur ").replace("÷", " bölü ").replace("=", " bərabərdir ").replace("°", " dərəcə ")
    text = re.sub(r"[*_#>~|]", "", text)
    return re.sub(r"\s+", " ", text).strip()


@api_router.post("/tts")
async def tts(req: TTSRequest):
    text = clean_for_tts(req.text)[:4000]
    if not text:
        raise HTTPException(400, "Mətn boşdur")
    speed = max(0.5, min(2.0, req.speed))
    voice = req.voice if req.voice in ("coral", "nova") else "coral"
    key = hashlib.sha256(f"{text}|{speed}|{voice}".encode()).hexdigest()
    cached = await db.tts_cache.find_one({"key": key})
    if cached and cached.get("storage_path"):
        try:
            audio = get_object(cached["storage_path"])
            return Response(content=audio, media_type="audio/mpeg",
                            headers={"Cache-Control": "public, max-age=31536000"})
        except Exception:
            pass
    audio = None
    try:
        engine = OpenAITextToSpeech(api_key=EMERGENT_KEY)
        audio = await engine.generate_speech(text=text, model="tts-1", voice=voice, speed=speed)
    except Exception as e:
        logger.error(f"TTS failed: {e}")
        raise HTTPException(500, "Səs yaradıla bilmədi")
    if not audio:
        raise HTTPException(500, "Səs yaradıla bilmədi")
    spath = f"{APP_NAME}/tts/{key}.mp3"
    try:
        put_object(spath, audio, "audio/mpeg")
        await db.tts_cache.insert_one({"key": key, "storage_path": spath})
    except Exception as e:
        logger.error(f"TTS cache store failed: {e}")
    return Response(content=audio, media_type="audio/mpeg",
                    headers={"Cache-Control": "public, max-age=31536000"})

const FONT_SIZES = ["20px", "24px", "28px"];

function App() {
  const [lang, setLang] = useState("az");
  const [theme, setTheme] = useState("light");
  const [fontStep, setFontStep] = useState(0);
  const [speed, setSpeed] = useState(1.0);
  const [voice, setVoice] = useState(() => localStorage.getItem("voice") || "coral");
  const [books, setBooks] = useState([]);
  const [subjects, setSubjects] = useState([]);
  const [view, setView] = useState({ type: "home" });

  const t = DICT[lang];

  useEffect(() => { document.documentElement.setAttribute("data-theme", theme); }, [theme]);
  useEffect(() => { document.documentElement.style.setProperty("--base-font", FONT_SIZES[fontStep]); }, [fontStep]);
  useEffect(() => { localStorage.setItem("voice", voice); }, [voice]);

  const loadBooks = useCallback(async () => {
    try {
      const res = await axios.get(`${API}/books`);
      setBooks(res.data);
    } catch (e) { console.error(e); }
  }, []);

  useEffect(() => {
    loadBooks();
    axios.get(`${API}/subjects`).then((r) => setSubjects(r.data)).catch(() => {});
  }, [loadBooks]);

  return (
    <div className="App">
      <AccessibilityTopBar {...{ t, lang, setLang, theme, setTheme, fontStep, setFontStep, speed, setSpeed, voice, setVoice, onOpenParent: () => setView({ type: "parent" }) }} />

          onChat={(ctx) => setView({ type: "chat", ctx, from: { type: "home" } })}
          onUploaded={loadBooks} />
      )}
      {view.type === "subject" && (
        <SubjectView t={t} subject={view.subject}

import { useState, useRef, useEffect, useCallback } from "react";

const Waveform = () => (

export const TutorChat = ({ t, lang, ctx, speed, voice, onBack }) => {
  const book = ctx.kind === "book" ? ctx.book : null;
  const subject = ctx.kind === "lesson" ? ctx.subject : null;
  const topic = ctx.kind === "lesson" ? ctx.topic : null;
  const [messages, setMessages] = useState([]);
  const [input, setInput] = useState("");
  const [streaming, setStreaming] = useState(false);
  const [listening, setListening] = useState(false);
  const [stage, setStage] = useState(null);
  const [retryAction, setRetryAction] = useState(null);
  const { speak, stop, speakingId, loading } = useTTS(lang, speed, voice);
  const scrollRef = useRef(null);
  const recRef = useRef(null);
  const inputRef = useRef(null);
  const autoSpokeRef = useRef(false);

  const scrollDown = () => setTimeout(() => scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: "smooth" }), 50);

  const qs = () => {
    const p = new URLSearchParams();
    if (book) p.set("book_id", book.id);
    if (subject) { p.set("subject", subject); p.set("topic", topic); }
    return p.toString();
  };

  const readStream = async (res, aiId) => {
    const reader = res.body.getReader();
    const decoder = new TextDecoder();
    let buffer = "", full = "", error = false;
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      buffer += decoder.decode(value, { stream: true });
      const parts = buffer.split("\n\n");
      buffer = parts.pop();
      for (const p of parts) {
        const line = p.replace(/^data: /, "").trim();
        if (!line) continue;
        try {
          const obj = JSON.parse(line);
          if (obj.delta) {
            full += obj.delta;
            setMessages((m) => m.map((x) => (x.id === aiId ? { ...x, content: full } : x)));
            scrollDown();
          } else if (obj.error) {
            error = true;
          }
        } catch (err) { console.warn("SSE parse skipped", err); }
      }
    }
    return { full, error };
  };

  const runAssistant = async (body, seedId, retryFn) => {
    const aiId = seedId || `a${Date.now()}`;
    setMessages((m) => (seedId ? [{ role: "assistant", content: "", id: aiId }] : [...m, { role: "assistant", content: "", id: aiId }]));
    setStreaming(true);
    setRetryAction(null);
    scrollDown();
    let result = { full: "", error: false };
    try {
      const res = await fetch(`${API}/chat`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(body),
      });
      if (!res.ok) throw new Error("http " + res.status);
      result = await readStream(res, aiId);
    } catch (err) {
      console.error("assistant error", err);
      result.error = true;
    } finally { setStreaming(false); scrollDown(); }
    if (result.error || !result.full.trim()) {
      setMessages((m) => m.map((x) => (x.id === aiId ? { ...x, content: "", error: true } : x)));
      setRetryAction(() => (retryFn || (() => runAssistant(body, seedId, retryFn))));
      return false;
    }
    speak(result.full, aiId);
    return true;
  };

  const streamExplain = async () => {
    const ok = await runAssistant({ subject, topic, lang, explain: true }, "explain", streamExplain);
    if (ok) setStage("explained");
  };

  const startRecall = () => { setStage("recall_input"); inputRef.current?.focus(); };

  const startQuiz = async () => {
    setStage("quiz");
    await runAssistant({ subject, topic, lang, stage: "quiz_next" }, undefined, startQuiz);
  };

  const initRef = useRef(null);
  useEffect(() => {
    const key = `${book?.id || ""}|${subject || ""}|${topic || ""}`;
    if (initRef.current === key) return;
    initRef.current = key;
    const load = async () => {
      if (subject && topic) { streamExplain(); return; }
      let hist = [];
      try {
        const res = await fetch(`${API}/messages?${qs()}`);
        hist = await res.json();
      } catch (err) { console.error("Failed to load messages", err); }
      if (hist.length > 0) { setMessages(hist); scrollDown(); return; }
      if (book) {
        setMessages([{ role: "assistant", content: book.explanation, id: "intro" }]);
        scrollDown();
      } else {
        setMessages([{ role: "assistant", content: t.welcome, id: "intro" }]);
        scrollDown();
      }
    };
    load();
    // eslint-disable-next-line
  }, [book?.id, subject, topic]);

  const send = useCallback(async (text) => {
    const msg = (text ?? input).trim();
    if (!msg || streaming) return;
    stop();
    setInput("");
    setRetryAction(null);
    const effStage = subject && topic
      ? (stage === "recall_input" ? "recall" : stage === "quiz" ? "quiz_answer" : undefined)
      : undefined;
    const userMsg = { role: "user", content: msg, id: `u${Date.now()}` };
    const aiId = `a${Date.now()}`;
    setMessages((m) => [...m, userMsg, { role: "assistant", content: "", id: aiId }]);
    setStreaming(true);
    scrollDown();
    const body = { book_id: book?.id || null, subject, topic, message: msg, lang, stage: effStage };
    let result = { full: "", error: false };
    try {
      const res = await fetch(`${API}/chat`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(body),
      });
      if (!res.ok) throw new Error("http " + res.status);
      result = await readStream(res, aiId);
    } catch (err) {
      console.error("send error", err);
      result.error = true;
    } finally {
      setStreaming(false);
      scrollDown();
    }
    if (result.error || !result.full.trim()) {
      setMessages((m) => m.map((x) => (x.id === aiId ? { ...x, content: "", error: true } : x)));
      setRetryAction(() => () => send(msg));
      return;
    }
    speak(result.full, aiId);
    if (effStage === "recall") setStage("recalled");
  }, [input, streaming, book, subject, topic, lang, stage, speak, stop]);

  const startMic = () => {
    const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!SR) { toast.error(t.micError); return; }
    stop();
    const rec = new SR();
    rec.lang = lang === "az" ? "az-AZ" : "tr-TR";
    rec.interimResults = false;
    rec.onresult = (e) => { const txt = e.results[0][0].transcript; setInput(txt); send(txt); };
    rec.onerror = () => { setListening(false); toast.error(t.micError); };
    rec.onend = () => setListening(false);
    recRef.current = rec;
    setListening(true);
    rec.start();
  };
  const stopMic = () => { recRef.current?.stop(); setListening(false); };

  const quicks = [t.q1, t.q2, t.q3];

  return (
    <div className="max-w-4xl mx-auto px-4 py-6 flex flex-col" style={{ height: "calc(100vh - 76px)" }}>
      {/* Header */}
      <div className="flex items-center gap-4 mb-4">
        <button data-testid="chat-back-btn" onClick={onBack}
          className="w-12 h-12 rounded-2xl bg-surface border-solid-2 flex items-center justify-center hover:bg-surface-2 transition-colors shrink-0">
          <ArrowLeft size={26} />
        </button>
        <div className="w-14 h-14 rounded-2xl bg-accent flex items-center justify-center shrink-0" style={{ color: "#fff" }}>
          <GraduationCap size={30} className="accent-fg" />
        </div>
        <div className="min-w-0">
          <h2 className="font-heading font-bold text-2xl truncate">{topic || (book ? book.title : t.general)}</h2>
          {subject && <p className="text-secondary text-base truncate">{subject}</p>}
          {book?.key_topics?.length > 0 && (
            <p className="text-secondary text-base truncate">{book.key_topics.join(" • ")}</p>
          )}
        </div>
      </div>

      {/* Messages */}
      <div ref={scrollRef} data-testid="chat-messages" className="flex-1 overflow-y-auto space-y-4 pr-1 pb-2">
        {messages.map((m) => (
          <div key={m.id} className={`rise flex ${m.role === "user" ? "justify-end" : "justify-start"}`}>
            <div className={`max-w-[85%] rounded-3xl px-5 py-4 border-solid-2 ${m.role === "user" ? "bg-accent text-white accent-fg" : "bg-surface"}`}
              style={m.role === "user" ? { borderColor: "var(--accent)" } : {}}>
              {m.error ? (
                <div data-testid="error-block">
                  <p className="text-xl leading-relaxed">{t.errorMsg}</p>
                  <button data-testid="retry-btn" onClick={() => { const fn = retryAction; setRetryAction(null); if (fn) fn(); }}
                    className="mt-3 inline-flex items-center gap-2 h-11 px-4 rounded-2xl bg-accent text-white accent-fg border-solid-2 font-bold text-lg hover:opacity-90 transition-opacity">
                    <RefreshCw size={22} /> {t.retryBtn}
                  </button>
                </div>
              ) : (
                <>
                  <p className="text-xl whitespace-pre-wrap leading-relaxed">{m.content || (streaming ? t.thinking : "")}</p>
                  {m.role === "assistant" && m.content && (
                    <button data-testid={`listen-btn-${m.id}`} onClick={() => (speakingId === m.id ? stop() : speak(m.content, m.id))}
                      className="mt-3 inline-flex items-center gap-2 h-11 px-4 rounded-2xl bg-surface-2 border-solid-2 font-bold text-lg hover:bg-accent hover:text-white transition-colors">
                      {speakingId === m.id ? (loading ? <Loader2 size={22} className="animate-spin" /> : <Square size={20} />) : <Volume2 size={22} />}
                      {speakingId === m.id ? t.stop : t.listen}
                    </button>
                  )}
                </>
              )}
            </div>
          </div>
        ))}
      </div>

      {/* Stage controls (lessons only) */}
      {subject && topic && stage && (
        <div data-testid="stage-indicator" className="flex flex-wrap items-center gap-2 text-base font-semibold mt-3">
          <span className={`px-3 py-1 rounded-full ${stage === "explained" ? "bg-accent text-white accent-fg" : "bg-surface-2 text-secondary"}`}>1. {t.stageExplain}</span>
          <span className={`px-3 py-1 rounded-full ${(stage === "recall_input" || stage === "recalled") ? "bg-accent text-white accent-fg" : "bg-surface-2 text-secondary"}`}>2. {t.stageRecall}</span>
          <span className={`px-3 py-1 rounded-full ${stage === "quiz" ? "bg-accent text-white accent-fg" : "bg-surface-2 text-secondary"}`}>3. {t.stageQuiz}</span>
        </div>
      )}
      {subject && topic && !streaming && (stage === "explained" || stage === "recalled") && (
        <div className="mt-3">
          {stage === "explained" && (
            <button data-testid="step-recall-btn" onClick={startRecall}
              className="w-full h-14 rounded-2xl bg-accent text-white accent-fg border-solid-2 font-heading font-bold text-xl flex items-center justify-center gap-3 hover:opacity-90 transition-opacity">

========== TTS HOOK ==========
import { useRef, useState, useCallback } from "react";

const BACKEND_URL = process.env.REACT_APP_BACKEND_URL;
const API = `${BACKEND_URL}/api`;

export function useTTS(lang, speed, voice) {
  const [speakingId, setSpeakingId] = useState(null);
  const [loading, setLoading] = useState(false);
  const audioRef = useRef(null);
  const tokenRef = useRef(0);

  const stop = useCallback(() => {
    tokenRef.current += 1;
    if (audioRef.current) {
      audioRef.current.pause();
      audioRef.current = null;
    }
    window.speechSynthesis?.cancel();
    setSpeakingId(null);
    setLoading(false);
  }, []);

  const speak = useCallback(async (text, id) => {
    stop();
    if (!text) return;
    const myToken = tokenRef.current;
    setSpeakingId(id);
    setLoading(true);
    try {
      const res = await fetch(`${API}/tts`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ text, lang, speed, voice }),
      });
      if (!res.ok) throw new Error("tts failed");
      const blob = await res.blob();
      if (myToken !== tokenRef.current) return;
      const url = URL.createObjectURL(blob);
      const audio = new Audio(url);
      audioRef.current = audio;
      audio.onended = () => {
        if (audioRef.current === audio) { audioRef.current = null; setSpeakingId(null); }
        URL.revokeObjectURL(url);
      };
      audio.onplay = () => setLoading(false);
      await audio.play();
    } catch (e) {
      if (myToken !== tokenRef.current) return;
      // Fallback to browser speech synthesis
      try {
        window.speechSynthesis.cancel();
        const u = new SpeechSynthesisUtterance(text);
        u.lang = lang === "az" ? "az-AZ" : "tr-TR";
        u.rate = speed;
        u.onend = () => setSpeakingId(null);
        setLoading(false);
        window.speechSynthesis.speak(u);
      } catch {
        setSpeakingId(null);
        setLoading(false);
      }
    }
  }, [lang, speed, voice, stop]);

  return { speak, stop, speakingId, loading };
}

    assert r.status_code == 200
    assert "text/event-stream" in r.headers.get("content-type", "")
    saw_delta = False
    saw_done = False
    for raw in r.iter_lines(decode_unicode=True):
        if not raw:
            continue
        if raw.startswith("data: "):
            payload = raw[6:]
            if '"delta"' in payload:
                saw_delta = True
            if '"done"' in payload:
                saw_done = True
                break
    assert saw_delta
    assert saw_done

## Original Problem Statement
Parent's child has weak eyesight and cannot read lessons. Build an AI system that scans PDF books, analyzes the topic, explains it simply to the child, does interactive Q&A, and helps solve problems — talking with the child interactively. Azerbaijani language support requested.

## User Choices
- Voice + visual together (TTS read-aloud + large text)
- Ask by typing AND speaking (microphone)
- LLM: Gemini 3.1 Pro (gemini-3.1-pro-preview)
- Level: middle school (11-14)
- No login (single user)
- Languages: Azerbaijani + Turkish

## Architecture
- Backend: FastAPI + MongoDB (motor). LLM via emergentintegrations (Gemini 3.1 Pro, streaming SSE). TTS via OpenAITextToSpeech (tts-1, voice coral). PDFs stored in Emergent Object Storage.
- Frontend: React 19 + Tailwind v4. Accessible, high-contrast, large-font UI (Lexend/Outfit). 3 themes (Parchment/Amber Night/Yellow-Black), font stepper, speed control, AZ/TR toggle.
- Voice input: browser Web Speech API (az-AZ/tr-TR).
